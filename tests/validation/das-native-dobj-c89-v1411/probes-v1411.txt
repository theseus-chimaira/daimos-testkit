probe v1411-native-c89-compile
command /OPTION/BASE/EXEC/KCC -Pgnu99 -O -x=pdp6 -m=gas -S -O /SCRATCH/V1411.A7 /CONFIG/V1403.C; ECHO COMPILE:$?
contains COMPILE:0
timeout 240
end
probe v1411-native-dobj-assembly
command /OPTION/BASE/EXEC/DAS -A -C -O /SCRATCH/V1411.DOBJ /SCRATCH/V1411.A7; ECHO ASSEMBLE:$?
contains ASSEMBLE:0
not_contains UUO
timeout 190
end
probe v1411-native-dobj-host-equality
command CMP /SCRATCH/V1411.DOBJ /CONFIG/V1411-HOST.DOBJ; ECHO OBJCMP:$?
contains OBJCMP:0
timeout 90
end
probe v1411-native-dlink-new-object
command /OPTION/BASE/EXEC/DLINK -O /SCRATCH/V1411.DXR -B 020 /CONFIG/V1403-CRT0.DOBJ /SCRATCH/V1411.DOBJ /CONFIG/V1403-SYSCALL.DOBJ; ECHO LINK:$?
contains LINK:0
not_contains UUO
timeout 240
end
probe v1411-native-dobj-c89-run
command /SCRATCH/V1411.DXR; ECHO RUNSTATUS:$?
contains RUNSTATUS:7
timeout 90
end
