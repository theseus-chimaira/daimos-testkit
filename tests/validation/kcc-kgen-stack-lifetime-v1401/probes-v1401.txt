# Each compiler invocation must return status zero; transcript alone is insufficient.
probe v1401-minimal-kcpp
command /OPTION/BASE/LIBEXEC/KCC/KCPP -R=/SCRATCH/V1401.MIN.KPT /CONFIG/V1373.C; ECHO MINCPP:$?
contains MINCPP:0
timeout 125
end
probe v1401-minimal-kparse
command /OPTION/BASE/LIBEXEC/KCC/KPARSE -R=/SCRATCH/V1401.MIN.KIR /SCRATCH/V1401.MIN.KPT; ECHO MINPARSE:$?
contains MINPARSE:0
timeout 210
end
probe v1401-minimal-kgen
command /OPTION/BASE/LIBEXEC/KCC/KGEN -R=/SCRATCH/V1401.MIN.KP2 /SCRATCH/V1401.MIN.KIR; ECHO MINGEN:$?
contains MINGEN:0
not_contains UUO
not_contains Internal error
timeout 210
end
probe v1401-minimal-kopt
command /OPTION/BASE/LIBEXEC/KCC/KOPT -R=/SCRATCH/V1401.MIN.A7 /SCRATCH/V1401.MIN.KP2; ECHO MINOPT:$?
contains MINOPT:0
timeout 125
end
probe v1401-minimal-equality

contains MINCMP:0
timeout 65
end
