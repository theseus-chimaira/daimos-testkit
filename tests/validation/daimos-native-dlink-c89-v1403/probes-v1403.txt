probe v1403-native-kcc
command /OPTION/BASE/EXEC/KCC -Pgnu99 -O -x=pdp6 -m=gas -S -O /SCRATCH/V1403-KCC.A7 /CONFIG/V1403.C; ECHO COMPILE:$?
contains COMPILE:0
not_contains UUO
timeout 240
end
probe v1403-native-das-compiled
command /OPTION/BASE/EXEC/DAS -A -O /SCRATCH/V1403-DAS.DXR /SCRATCH/V1403-KCC.A7; ECHO ASSEMBLE:$?
contains ASSEMBLE:0
timeout 190
end
probe v1403-native-dlink

contains LINK:0
not_contains UUO
timeout 240
end

end
