# Native DAS packed ASCII7 input and unchanged default S6REC regression.
probe v1402-ascii7
command /OPTION/BASE/EXEC/DAS -A -O /SCRATCH/V1402-A.DXR /CONFIG/V1402-SRC.A7; ECHO A7STATUS:$?
contains A7STATUS:0
not_contains UUO
timeout 140
end
probe v1402-ascii7-equality
command CMP /SCRATCH/V1402-A.DXR /CONFIG/V1402-GOLD.DXR; ECHO A7CMP:$?
contains A7CMP:0
timeout 65
end
