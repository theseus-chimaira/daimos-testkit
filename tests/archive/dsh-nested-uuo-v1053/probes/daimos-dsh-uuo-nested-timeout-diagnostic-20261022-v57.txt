# Exact 34-probe history preceding the first ECHO PIPEOK ! CAT failure.
# Compare only against explicitly verified installed DSH DXR bytes.
# Path/filesystem and UUO failures are both considered regressions.
probe dsh-comment-in-word
command ECHO A#B
line ^A#B$
end

probe dsh-empty-argument
command ECHO A "" B
line ^A  B$
end

probe dsh-quoted-keyword
command ECHO "IF"
line ^IF$
end

probe dsh-single-quote-backslash
command ECHO 'A\B'
line ^A\\B$
end

probe dsh-case-wildcard
command CASE FOO IN WHEN B* DO ECHO BAD WHEN F?O DO ECHO CASEOK ESAC
line ^CASEOK$
not_contains BAD
end

probe dsh-case-quoted-wildcard
command CASE FOO IN WHEN "F?O" DO ECHO BAD WHEN F* DO ECHO CASEQUOTEOK ESAC
line ^CASEQUOTEOK$
not_contains BAD
end

probe dsh-glob-create
setup ECHO ONE > /SYSTEM/DSHG1
setup ECHO TWO > /SYSTEM/DSHG2
command ECHO /SYSTEM/DSHG*
contains /SYSTEM/DSHG1
contains /SYSTEM/DSHG2
end

probe dsh-glob-quoted
command ECHO "/SYSTEM/DSHG*"
line ^/SYSTEM/DSHG\*$
end

probe dsh-glob-unmatched
command ECHO /SYSTEM/NOMATCH*
line ^/SYSTEM/NOMATCH\*$
end

probe dsh-glob-variable-unquoted
setup SET P "/SYSTEM/DSHG*"
command ECHO $P
contains /SYSTEM/DSHG1
contains /SYSTEM/DSHG2
end

probe dsh-glob-variable-quoted
setup SET P "/SYSTEM/DSHG*"
command ECHO "$P"
line ^/SYSTEM/DSHG\*$
end

probe dsh-glob-cleanup-one
command RM /SYSTEM/DSHG1
not_contains RM:
end

probe dsh-glob-cleanup-two
command RM /SYSTEM/DSHG2
not_contains RM:
end

end

probe dsh-variable-bracketed

end

probe dsh-variable-resize-unset
setup SET KEEP STABLE
setup SET GROW A
setup SET GROW ABCDEFGHIJKLMNOPQRSTUVWXYZ
setup UNSET V10
setup SET AFTER REUSED
command ECHO $KEEP; ECHO $GROW; ECHO $AFTER; ECHO X$V10
contains STABLE
contains ABCDEFGHIJKLMNOPQRSTUVWXYZ
contains REUSED
line ^X$
end

# Repeated native process-group launches after the full lookup history.
# These are deliberately separate multi-stage stress cases so a kernel fault
# cannot be silently misattributed to a particular GCC source revision.
probe dsh-uuo-pipeline-loop-two-stage-20261022-v3
command FOR I IN A B C D E F G H DO ECHO $I ! CAT DONE
line ^A$
line ^H$
end

probe dsh-uuo-pipeline-loop-three-stage-20261022-v3
command FOR I IN J K L M N O P Q DO ECHO $I ! CAT ! CAT DONE
line ^J$
line ^Q$
end

probe dsh-uuo-pipeline-after-loops-20261022-v3
command ECHO PIPEFINAL ! CAT; ECHO STATUS:$?
line ^PIPEFINAL$
line ^STATUS:0$
end

# Increase the number of RUN/WAIΤ cycles within a single shell process.
probe dsh-uuo-nested-external-launch-20261022-v4
command FOR I IN A B C D DO FOR J IN 1 2 3 4 5 6 DO ECHO $I$J DONE DONE
line ^A1$
line ^D6$
end

probe dsh-uuo-nested-two-stage-20261022-v4
command FOR I IN A B C D DO FOR J IN 1 2 3 4 5 6 DO ECHO $I$J ! CAT DONE DONE
line ^A1$
line ^D6$
end

probe dsh-uuo-nested-three-stage-20261022-v4
command FOR I IN A B C D DO ECHO $I ! CAT ! CAT ! CAT DONE
line ^A$
line ^D$
timeout 60
end

probe dsh-uuo-after-heavy-launch-20261022-v4
command ECHO LAUNCHOK; ECHO PIPEOK ! CAT
line ^LAUNCHOK$
line ^PIPEOK$
end
