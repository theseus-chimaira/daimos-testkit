# Use an isolated writable D6FS root image and --login ROOT.
probe append-initial
command ECHO FIRST > /CONFIG/APPENDTEST
end
probe append-second
command ECHO SECOND >> /CONFIG/APPENDTEST
end
probe append-third
command ECHO THIRD >> /CONFIG/APPENDTEST
end
probe append-readback
command CAT /CONFIG/APPENDTEST
contains FIRST
contains SECOND
contains THIRD
end
probe orderly-halt
command HALT
terminal eof
end
